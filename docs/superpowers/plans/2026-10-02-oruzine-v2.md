```typescript
// apps/web/src/stores/fontStore.ts
import { get, set } from 'idb-keyval';
import { create } from 'zustand';

const GOOGLE_FONTS = [
  'Special Elite', 'Black Ops One', 'VT323', 'Bungee Shade',
  'Permanent Marker', 'Syne Mono', 'Cinzel Decorative',
];

interface CustomFontEntry {
  family: string;
  dataUrl: string;
  format: 'woff2' | 'truetype' | 'opentype';
}

interface FontStoreState {
  googleFontsLoaded: boolean;
  customFonts: CustomFontEntry[];
  allFamilies: string[];
  loadGoogleFonts: () => Promise<void>;
  uploadFont: (file: File) => Promise<string>;
  getFontDataUrl: (family: string) => string | null;
  loadCustomFontsFromIdb: () => Promise<void>;
}

export const useFontStore = create<FontStoreState>((set, get) => ({
  googleFontsLoaded: false,
  customFonts: [],
  allFamilies: [...GOOGLE_FONTS],

  loadGoogleFonts: async () => {
    if (get().googleFontsLoaded) return;
    const families = GOOGLE_FONTS.map((f) => f.replace(/ /g, '+')).join('&family=');
    const link = document.createElement('link');
    link.rel = 'stylesheet';
    link.href = `https://fonts.googleapis.com/css2?family=${families}&display=swap`;
    document.head.appendChild(link);
    set({ googleFontsLoaded: true });
  },

  uploadFont: async (file: File) => {
    const buffer = await file.arrayBuffer();
    const ext = file.name.split('.').pop()?.toLowerCase() ?? 'ttf';
    const format = ext === 'woff2' ? 'woff2' : ext === 'otf' ? 'opentype' : 'truetype';
    const family = file.name.replace(/\.(ttf|otf|woff2)$/i, '').replace(/[^a-zA-Z0-9 ]/g, ' ');

    const fontFace = new FontFace(family, buffer);
    await fontFace.load();
    document.fonts.add(fontFace);

    const blob = new Blob([buffer]);
    const reader = new FileReader();
    const dataUrl = await new Promise<string>((resolve) => {
      reader.onload = () => resolve(reader.result as string);
      reader.readAsDataURL(blob);
    });

    const entry: CustomFontEntry = { family, dataUrl, format };
    const current = get().customFonts;
    const updated = [...current.filter((f) => f.family !== family), entry];

    await set_idb('oruzine-custom-fonts', updated);
    set({
      customFonts: updated,
      allFamilies: [...GOOGLE_FONTS, ...updated.map((f) => f.family)],
    });

    return family;
  },

  getFontDataUrl: (family: string) => {
    return get().customFonts.find((f) => f.family === family)?.dataUrl ?? null;
  },

  loadCustomFontsFromIdb: async () => {
    const stored = await get_idb<CustomFontEntry[]>('oruzine-custom-fonts');
    if (!stored || stored.length === 0) return;

    for (const entry of stored) {
      try {
        const response = await fetch(entry.dataUrl);
        const buffer = await response.arrayBuffer();
        const fontFace = new FontFace(entry.family, buffer);
        await fontFace.load();
        document.fonts.add(fontFace);
      } catch { /* skip broken fonts */ }
    }

    set({
      customFonts: stored,
      allFamilies: [...GOOGLE_FONTS, ...stored.map((f) => f.family)],
    });
  },
}));

// Wrap idb-keyval to avoid name collision with zustand's get
async function get_idb<T>(key: string): Promise<T | undefined> {
  const { get: idbGet } = await import('idb-keyval');
  return idbGet<T>(key);
}
async function set_idb(key: string, val: unknown): Promise<void> {
  const { set: idbSet } = await import('idb-keyval');
  return idbSet(key, val);
}
```

```tsx
// apps/web/src/features/editor/StyledTextRenderer.tsx
import React, { useMemo } from 'react';
import type { TextObject } from '../../stores/documentStore';

interface Props {
  text: TextObject;
  containerWidth: number;
  containerHeight: number;
  fontScale?: number; // For miniature preview scaling
}

export const StyledTextRenderer: React.FC<Props> = ({
  text: t,
  containerWidth: cw,
  containerHeight: ch,
  fontScale = 1,
}) => {
  const fontSize = t.fontSizePt * fontScale;
  const hasWarp = t.warpType && t.warpType !== 'none';
  const hasStroke = (t.strokeWidth ?? 0) > 0;
  const hasShadow = t.shadowStyle && t.shadowStyle !== 'none';
  const hasBadge = !!t.badgeFill;

  const curvePath = useMemo(() => {
    if (!hasWarp) return '';
    const amount = (t.warpAmount ?? 50) / 100;
    const w = cw;
    const h = ch;
    const cy = h / 2;

    switch (t.warpType) {
      case 'arc-up': {
        const bend = -amount * h * 0.6;
        return `M 0,${cy} Q ${w / 2},${cy + bend} ${w},${cy}`;
      }
      case 'arc-down': {
        const bend = amount * h * 0.6;
        return `M 0,${cy} Q ${w / 2},${cy + bend} ${w},${cy}`;
      }
      case 'wave': {
        const amp = amount * h * 0.3;
        return `M 0,${cy} C ${w * 0.25},${cy - amp} ${w * 0.75},${cy + amp} ${w},${cy}`;
      }
      case 'badge-circle': {
        const r = Math.min(w, h) * 0.45;
        const cx = w / 2;
        return `M ${cx - r},${cy} A ${r},${r} 0 1,1 ${cx + r},${cy} A ${r},${r} 0 1,1 ${cx - r},${cy}`;
      }
      default:
        return '';
    }
  }, [hasWarp, t.warpType, t.warpAmount, cw, ch]);

  const pathId = useMemo(
    () => `warp-${t.id}-${Math.random().toString(36).slice(2, 8)}`,
    [t.id],
  );

  const shadowLayers = useMemo(() => {
    if (!hasShadow) return [];
    const color = t.shadowColor ?? 'rgba(0,0,0,0.5)';
    const ox = t.shadowOffset?.x ?? 2;
    const oy = t.shadowOffset?.y ?? 2;

    switch (t.shadowStyle) {
      case 'drop':
        return [{ dx: ox, dy: oy, color }];
      case 'retro-hard':
        return [{ dx: ox * 2, dy: oy * 2, color }];
      case 'extruded-3d': {
        const layers = [];
        for (let i = 1; i <= 4; i++) {
          layers.push({ dx: ox * i * 0.5, dy: oy * i * 0.5, color });
        }
        return layers;
      }
      default:
        return [];
    }
  }, [hasShadow, t.shadowStyle, t.shadowColor, t.shadowOffset]);

  const textStyle: React.CSSProperties = {
    fontFamily: t.fontFamily ?? 'sans-serif',
    fontSize: `${fontSize}px`,
    fontWeight: t.bold ? 'bold' : 'normal',
    fontStyle: t.italic ? 'italic' : 'normal',
    letterSpacing: t.letterSpacing ? `${t.letterSpacing}px` : undefined,
  };

  const renderTextContent = (dx = 0, dy = 0, fill: string = t.color) => {
    const commonProps = {
      fill,
      stroke: hasStroke ? (t.strokeColor ?? '#000') : undefined,
      strokeWidth: hasStroke ? t.strokeWidth : undefined,
      paintOrder: hasStroke ? 'stroke' : undefined,
      style: textStyle,
      textAnchor: t.align === 'right' ? 'end' : t.align === 'center' ? 'middle' : 'start',
      dominantBaseline: 'central' as const,
    };

    if (hasWarp && curvePath) {
      return (
        <text {...commonProps} dx={dx} dy={dy}>
          <textPath href={`#${pathId}`} startOffset="50%" textAnchor="middle">
            {t.text}
          </textPath>
        </text>
      );
    }

    const x = t.align === 'right' ? cw : t.align === 'center' ? cw / 2 : 0;
    return (
      <text {...commonProps} x={x + dx} y={ch / 2 + dy}>
        {t.text}
      </text>
    );
  };

  return (
    <svg
      width={cw}
      height={ch}
      viewBox={`0 0 ${cw} ${ch}`}
      xmlns="http://www.w3.org/2000/svg"
      overflow="visible"
    >
      {hasWarp && curvePath && (
        <defs>
          <path id={pathId} d={curvePath} fill="none" />
        </defs>
      )}

      {hasBadge && (
        <rect
          x={0}
          y={0}
          width={cw}
          height={ch}
          rx={t.badgeRadius ?? 0}
          fill={t.badgeFill}
        />
      )}

      {shadowLayers.map((layer, i) => (
        <React.Fragment key={`shadow-${i}`}>
          {renderTextContent(layer.dx, layer.dy, layer.color)}
        </React.Fragment>
      ))}

      {renderTextContent()}
    </svg>
  );
};
```

```typescript
// apps/web/src/hooks/useTouchGestures.ts
import { useEffect, useRef, type RefObject } from 'react';
import Hammer from 'hammerjs';

interface TouchGestureHandlers {
  onPinchZoom?: (scale: number, centerX: number, centerY: number) => void;
  onPan?: (deltaX: number, deltaY: number) => void;
  onRotate?: (angleDeg: number) => void;
  onDoubleTap?: (x: number, y: number) => void;
  onLongPress?: (x: number, y: number) => void;
}

export function useTouchGestures(
  elementRef: RefObject<HTMLElement | null>,
  handlers: TouchGestureHandlers,
) {
  const hammerRef = useRef<HammerManager | null>(null);

  useEffect(() => {
    const el = elementRef.current;
    if (!el) return;

    const mc = new Hammer.Manager(el, { touchAction: 'none' });
    mc.add(new Hammer.Pinch({ enable: true }));
    mc.add(new Hammer.Pan({ direction: Hammer.DIRECTION_ALL, threshold: 5 }));
    mc.add(new Hammer.Rotate({ enable: true }));
    mc.add(new Hammer.Tap({ event: 'doubletap', taps: 2 }));
    mc.add(new Hammer.Press({ time: 500 }));

    mc.on('pinch', (e) => handlers.onPinchZoom?.(e.scale, e.center.x, e.center.y));
    mc.on('pan', (e) => handlers.onPan?.(e.deltaX, e.deltaY));
    mc.on('rotate', (e) => handlers.onRotate?.(e.rotation));
    mc.on('doubletap', (e) => handlers.onDoubleTap?.(e.center.x, e.center.y));
    mc.on('press', (e) => handlers.onLongPress?.(e.center.x, e.center.y));

    hammerRef.current = mc;
    return () => { mc.destroy(); hammerRef.current = null; };
  }, [elementRef, handlers]);
}
```

```typescript
export async function detectWebGPU(): Promise<boolean> {
  try {
    if (typeof navigator !== 'undefined' && 'gpu' in navigator) {
      const adapter = await (navigator as any).gpu.requestAdapter();
      return !!adapter;
    }
  } catch {}
  return false;
}

export function getExecutionProviders(): string[] {
  // Will be resolved at runtime
  return ['webgpu', 'wasm'];
```

```typescript
import { pipeline, env } from '@huggingface/transformers';

env.allowLocalModels = false;

let segmenter: any = null;

export async function removeBackgroundAI(
  imageSource: string | HTMLImageElement,
  onProgress?: (percent: number, message: string) => void,
): Promise<string> {
  onProgress?.(10, 'Loading AI model (first time may take a moment)...');

  if (!segmenter) {
    segmenter = await pipeline('image-segmentation', 'onnx-community/ormbg-ONNX', {
      dtype: 'q8',
      device: 'webgpu',
    });
  }

  onProgress?.(50, 'Running AI segmentation...');
  const src = typeof imageSource === 'string' ? imageSource : imageSource.src;
  const output = await segmenter(src);

  onProgress?.(90, 'Compositing cutout...');
  // Extract the mask and composite
  const maskUrl = output[0]?.mask ?? output[0]?.url;

  onProgress?.(100, 'Done');
  return maskUrl;
}
```
